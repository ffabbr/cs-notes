>         double[] g = new double[n + 1];

>         }

>         }

> 
>         for (int i = n; i >= 1; i--) {
>             dp[i] = p[i] * (g[i] + dp[i + 1]) + (1.0 - p[i]) * dp[i + 2];
>         }

>         Out.println(dp[1]);
>     }
> }
